class Solution {
public:
    int removeDuplicates(vector<int>& nums) {

        if (nums.empty())
            return 0;

        int unique = 0;

        for (int officer = 1; officer < nums.size(); officer++) {

            if (nums[officer] != nums[unique]) {
                unique++;
                nums[unique] = nums[officer];
            }
        }
