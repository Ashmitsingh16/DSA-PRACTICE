int removeDuplicates(vector<int>& nums) {

        int unique = 0;

        for (int officer = 0; officer < nums.size(); officer++) {

            if (unique < 2 || nums[officer] != nums[unique - 2]) {

                nums[unique] = nums[officer];
                unique++;
            }
        }

        return unique;
    }
};
